def calc_avg(grades, class_name=None):
    """Calculate the average grade for a(all) class(es)."""
    if class_name is None or class_name == "":
        mean = {}
        for class_name_i in grades:
            mean[class_name_i] = round(
                statistics.mean(grades[class_name_i].values()), 2
            )
        print("Average grade calculated for all classes.")
    elif class_name not in grades:
        print(f"Class {class_name} not found.")
        return
    else:
        mean = round(statistics.mean(grades[class_name].values()), 2)
        print(f"Average grade calculated for class {class_name}.")
    return mean


def calc_min_max(grades, class_name=None):
    """Calculate the min and max grades for a(all) class(es)."""
    if class_name is None or class_name == "":
        minmax = {}
        for class_name_i in grades:
            minmax[class_name_i] = [
                min(grades[class_name_i].values()),
                max(grades[class_name_i].values()),
            ]
        print("Min and max grades calculated for all classes.")
    elif class_name not in grades:
        print(f"Class {class_name} not found.")
        return
    else:
        minmax = [min(grades[class_name].values()), max(grades[class_name].values())]
        print(f"Min and max grades calculated for class {class_name}.")
    return minmax


def grade_to_letter(grade):
    """Calculate the letter grade corresponding to the numeric grade."""
    if grade >= 90:
        return "A"
    elif grade >= 80:
        return "B"
    elif grade >= 70:
        return "C"
    elif grade >= 60:
        return "D"
    else:
        return "F"


def count_grades(letters):
    """Count the distribution of letter grades for a(all) class(es)."""
    return sorted(Counter(letters).items())


def letter_grades(grades, class_name=None):
    """Calculate the of letter grades for a(all) class(es)."""
    if class_name is None or class_name == "":
        letters = {}
        for class_name_i in grades:
            letters[class_name_i] = [
                grade_to_letter(grd) for grd in grades[class_name_i].values()
            ]
    else:
        letters = []
        for grd in grades[class_name].values():
            letters.append(grade_to_letter(grd))
    return letters


def letter_distribution(grades, class_name=None):
    """Calculate the distribution of letter grades for a(all) class(es)."""
    if class_name is None or class_name == "" or class_name in grades:
        letters = letter_grades(grades, class_name)
    else:
        print(f"Class {class_name} not found.")
    if class_name is None or class_name == "":
        counts = {}
        for class_name_i in letters:
            counts[class_name_i] = count_grades(letters[class_name_i])
        print("Letter grade distribution calculated for all classes.")
    elif class_name not in grades:
        return
    else:
        counts = count_grades(letters)
        print(f"Letter grade distribution calculated for class {class_name}.")
    return counts

def print_class(grades, class_name):
    """Print all students' grades for an existing class"""
    if class_name not in grades:
        print(f"Class {class_name} not found.")
    else:
        pp(grades[class_name])
        print(f"Gradebook printed for class {class_name}.")


def change_grade(grades, class_name, student, new_grade):
    """Update a student's grade after checking for existing class / student and valid grade"""
    if class_name not in grades:
        print(f"Class {class_name} not found.")
    elif student not in grades[class_name]:
        print(f"Student {student} not found in class {class_name}.")
    elif not MIN_GRADE <= new_grade <= MAX_GRADE:
        print(f"New grade of {new_grade} is not valid.")
    else:
        grades[class_name][student] = new_grade
        print(f"Grade for student {student} in class {class_name} changed.")


def add_student(grades, class_name, student, grade):
    """Add a new student to an existing class and set grade"""
    if class_name not in grades:
        print(f"class {class_name} not found")
    elif student in grades[class_name]:
        print(f"student {student} is already in class {class_name}")
    elif not 0 <= grade <= 100:
        print(f"new grade of {grade} is not valid")
    else:
        grades[class_name][student] = grade
        print(f"Student {student} added in class {class_name}.")


def remove_student(grades, class_name, student):
    """Remove an existing student from an existing class"""
    if class_name not in grades:
        print(f"class {class_name} not found")
    elif student not in grades[class_name]:
        print(f"student {student} not found in class {class_name}")
    else:
        del grades[class_name][student]
        print(f"Student {student} removed from class {class_name}.")


def add_class(grades, class_name):
    """Add a new class to the gradebook"""
    if class_name in grades:
        print(f"class {class_name} is already in gradebook")
    else:
        grades[class_name] = {}
        print(f"Class {class_name} added to gradebook.")

def generate_gradebook(classes):
    """Generate gradebook for X number of classes, each with a random number of student, grade pairs"""
    grades = {}
    for class_name in classes:
        n_students = random.randint(a=10, b=25)
        names = [
            "".join(random.choices(population=string.ascii_uppercase, k=2))
            for _ in range(n_students)
        ]
        grds = np.round(
            np.random.normal(loc=random.randint(a=70, b=80), scale=5, size=n_students),
            1,
        )
        grades[class_name] = dict(zip(names, grds))
    return grades


def save_gradebook(grades, file_name):
    # with open(filename, mode="w") as file: # manually write csv file
    #     file.write("class_name,student,grade\n")
    #     for class_name, students in grades.items():
    #         for student, grade in students.items():
    #             file.write(f"{class_name},{student},{grade}\n")
    with open(file_name, mode="w", newline="") as file:  # now using csv
        writer = csv.writer(file)
        writer.writerow(["class", "student", "grade"])
        for class_name, students in grades.items():
            for student, grade in students.items():
                writer.writerow([class_name, student, grade])
    print(f"Gradebook saved as {file_name}.")


def load_gradebook(file_name):
    """Load gradebook and output as a nested dictionary"""
    with open(file_name, "r", newline="") as file:
        grades = {}
        reader = csv.reader(file)
        next(reader)
        for row in reader:
            class_name = row[0]
            student = row[1]
            grade = float(row[2])
            if class_name not in grades:
                grades[class_name] = {}
            grades[class_name][student] = grade
    print(f"Gradebook {file_name} loaded.")
    return grades
