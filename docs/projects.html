Analyze gradebook

# ---------------------------------------
# analyze gradebook
# ---------------------------------------
# define functions
def print_class(grades, cls):
    """Print the grades of each student in the class."""
    pp(grades[cls])


def calc_avg(grades, cls=None):
    """Calculate the average grade for a(all) class(es)."""
    if cls is None:
        mean = {}
        for cls_i in grades:
            mean[cls_i] = round(statistics.mean(grades[cls_i].values()), 2)
    else:
        mean = round(statistics.mean(grades[cls].values()), 2)
    return mean


def calc_min_max(grades, cls=None):
    """Calculate the min and max grades for a(all) class(es)."""
    if cls is None:
        minmax = {}
        for cls_i in grades:
            minmax[cls_i] = [min(grades[cls_i].values()), max(grades[cls_i].values())]
    else:
        minmax = [min(grades[cls].values()), max(grades[cls].values())]
    return minmax


def grade_to_letter(grade):
    """Calculate the letter grade corresponding to the numeric grade."""
    if grade >= 90:
        return "A"
    elif grade >= 80:
        return "B"
    elif grade >= 70:
        return "C"
    elif grade >= 60:
        return "D"
    else:
        return "F"


def count_grades(letters):
    """Count the distribution of letter grades for a(all) class(es)."""
    return sorted(Counter(letters).items())


def letter_grades(grades, cls=None):
    """Calculate the of letter grades for a(all) class(es)."""
    if cls is None:
        letters = {}
        for cls_i in grades:
            letters[cls_i] = [grade_to_letter(grd) for grd in grades[cls_i].values()]
    else:
        letters = []
        for grd in grades[cls].values():
            letters.append(grade_to_letter(grd))
    return letters


def letter_distribution(grades, cls=None):
    """Calculate the distribution of letter grades for a(all) class(es)."""
    letters = letter_grades(grades, cls)

    if cls is None:
        counts = {}
        for cls_i in letters:
            counts[cls_i] = count_grades(letters[cls_i])
    else:
        counts = count_grades(letters)
    return counts
